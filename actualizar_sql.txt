-- =====================================================================
-- Proceso Ventas y Compras — Cuentas contables para asientos automáticos
-- gestion_multipyme | v2.0 | 2026-09-25
-- =====================================================================
-- CAMBIO DE ENFOQUE RESPECTO A v1.0:
--   v1.0 resolvía la cuenta de ingreso por CASCADA producto -> categoria
--   -> general, calculada en tiempo de ejecucion del motor de asientos.
--   v2.0 la reemplaza por un modelo de GRANULARIDAD POR PRODUCTO: cada
--   producto guarda sus propias cuentas de compra, venta y CMV. El
--   valor default ya no se calcula en cada asiento: se MATERIALIZA en
--   el producto al momento del alta (mismo patron de materializacion
--   ya usado en el sistema, ej. saldo_pendiente). La categoria pasa a
--   ser una herramienta opcional de carga masiva / semilla, no un
--   participante del motor de asientos en runtime.
--
-- Alcance de esta entrega — asientos automaticos cubiertos:
--   1) Compra de producto para reventa (con o sin control de stock)
--      DEBE  producto.cont_cuenta_compra_id      (131 Mercaderias)
--      DEBE  iva_alicuotas.cont_cuenta_compra_id  (IVA Credito Fiscal)
--      HABER entidad.cont_cuenta_id_proveedor
--   2) Venta de producto con stock
--      DEBE  entidad.cont_cuenta_id_cliente
--      HABER producto.cont_cuenta_venta_id        (600 Ventas Productos)
--      HABER iva_alicuotas.cont_cuenta_venta_id    (IVA Debito Fiscal)
--      (diferido, fuera de alcance) CMV: producto.cont_cuenta_cmv_id / 131
--   3) Venta de producto sin stock: igual a (2) sin linea de CMV
--   4) Venta de servicios: igual a (2)/(3) contra 601 Ventas Servicios,
--      nunca CMV. Diferencia de cambio (611/842) queda FUERA del modelo
--      de producto — depende de moneda_id/tipo_cambio del comprobante,
--      no del producto vendido.
--   5) Compra de producto sin stock: igual a (1), va a Mercaderias (131)
--      igual que con stock — controla_stock solo determina si hay
--      movimiento de inventario, no cambia la cuenta contable.
--   6) Compra de servicios: DEBE cuenta de GASTO via
--      producto.cont_cuenta_compra_id (no 131), + IVA Credito si aplica.
--
-- Fuera de alcance en esta entrega:
--   - CMV en el asiento automatico de venta (se sigue registrando manual
--     a fin de mes; cont_cuenta_cmv_id queda preparado como provision).
--   - Diferencia de cambio, retenciones y percepciones: cuentas fijas
--     resueltas por otra via (impuestos_tipos.cuenta_contable_id y
--     cuentas fijas 611/842), no por el modelo de producto.
--   - Backfill masivo de los ~6.000 productos existentes sin cuenta
--     cargada: se trata en script aparte, no en este DDL.
-- =====================================================================

START TRANSACTION;

-- ---------------------------------------------------------------------
-- 1) IVA: separar cuenta de compra (credito fiscal) de cuenta de venta
--    (debito fiscal). Hoy `iva_alicuotas` tiene un solo campo y apunta
--    a la cuenta de credito fiscal (845/844) — sirve solo para compras.
--    (Sin cambios respecto a v1.0.)
-- ---------------------------------------------------------------------
ALTER TABLE gestion__impuestos__iva_alicuotas
  CHANGE COLUMN cont_cuenta_id cont_cuenta_compra_id INT(11) NOT NULL DEFAULT 0
    COMMENT 'IVA Credito Fiscal (compras) — antes cont_cuenta_id',
  ADD COLUMN cont_cuenta_venta_id INT(11) NOT NULL DEFAULT 0
    COMMENT 'IVA Debito Fiscal (ventas)'
    AFTER cont_cuenta_compra_id;

-- Completar cont_cuenta_venta_id por alicuota real antes de activar el
-- motor de ventas. Ejemplo (AJUSTAR segun iva_alicuota_id reales):
-- UPDATE gestion__impuestos__iva_alicuotas
-- SET cont_cuenta_venta_id = 331  -- IVA Debito Fiscal
-- WHERE iva_alicuota_id IN (...);

-- ---------------------------------------------------------------------
-- 2) Origen de asiento por subgrupo de comprobante.
--    (Sin cambios respecto a v1.0.)
-- ---------------------------------------------------------------------
ALTER TABLE gestion__comprobantes_subgrupos
  ADD COLUMN origen_asiento ENUM('AUTO','MANU','NINGUNO') NOT NULL DEFAULT 'NINGUNO'
    AFTER tipo_asiento_id;

UPDATE gestion__comprobantes_subgrupos
SET origen_asiento = 'AUTO'
WHERE comprobante_subgrupo_id = 1; -- Facturacion (confirmar id real en produccion)

-- ---------------------------------------------------------------------
-- 3) BAJA del modelo por categoria de v1.0.
--    Si el DDL v1.0 ya se corrio en produccion, revertir estos campos
--    de gestion__productos_categorias (la categoria deja de participar
--    del motor de asientos en runtime):
-- ---------------------------------------------------------------------
-- ALTER TABLE gestion__productos_categorias
--   DROP FOREIGN KEY fk_categoria_producto_tipo,
--   DROP KEY uk_categoria_tipo,
--   DROP COLUMN cont_cuenta_ingreso_id,
--   DROP COLUMN cont_cuenta_cmv_id;
--   -- producto_tipo_id se conserva: sigue siendo util como semilla para
--   -- carga masiva (ver paso 5), solo deja de ser FK obligatoria.

-- Si v1.0 tambien llego a correr el paso 4 original (FK compuesta
-- producto-categoria), revertir primero en gestion__productos:
-- ALTER TABLE gestion__productos
--   DROP FOREIGN KEY fk_producto_categoria_tipo,
--   DROP COLUMN cont_cuenta_ingreso_id; -- se reemplaza por cont_cuenta_venta_id (paso 4)

-- ---------------------------------------------------------------------
-- 4) Modelo por PRODUCTO: tres cuentas propias (compra, venta, CMV) +
--    dos flags independientes (aplica_compra, aplica_venta).
--    NULL en cualquier cuenta = pendiente de backfill (no "no aplica";
--    "no aplica" se expresa con el flag correspondiente en 0).
-- ---------------------------------------------------------------------
ALTER TABLE gestion__productos
  ADD COLUMN aplica_compra TINYINT(1) NOT NULL DEFAULT 1
    COMMENT '1 = el producto puede comprarse',
  ADD COLUMN aplica_venta TINYINT(1) NOT NULL DEFAULT 1
    COMMENT '1 = el producto puede venderse',
  ADD COLUMN cont_cuenta_compra_id INT(11) DEFAULT NULL
    COMMENT 'Cuenta de compra: 131 Mercaderias si es bien de cambio (con o sin control de stock), o cuenta de gasto si es servicio',
  ADD COLUMN cont_cuenta_venta_id INT(11) DEFAULT NULL
    COMMENT 'Cuenta de ingreso: 600 Ventas Productos o 601 Ventas Servicios segun corresponda',
  ADD COLUMN cont_cuenta_cmv_id INT(11) DEFAULT NULL
    COMMENT 'Cuenta de costo de mercaderia vendida — solo se completa y solo se usa si controla_stock = 1. Provision: no usada por el motor de asientos en esta entrega';

-- Nota de diseno: el gate del CMV en el motor de asientos es
-- `controla_stock`, NO `es_servicio`. Confirmado por el caso de un
-- producto fisico sin control de stock: se vende igual que un producto
-- con stock (misma cuenta de venta) pero sin generar linea de CMV.

-- ---------------------------------------------------------------------
-- 5) Semilla al alta: la cuenta default de un producto nuevo se
--    materializa (se copia) desde gestion__productos_tipos al momento
--    de crear el producto, segun corresponda a producto_tipo_id
--    (Producto -> 131 compra / 600 venta; Servicio -> cuenta de gasto
--    compra / 601 venta). Esto reemplaza la resolucion en cascada por
--    categoria de v1.0: la categoria puede seguir usandose como agrupador
--    visual o para carga masiva asistida, pero no la lee el motor de
--    asientos en runtime.
--    Se agregan las cuentas semilla a nivel tipo de producto para que
--    el alta tenga de donde copiar:
-- ---------------------------------------------------------------------
ALTER TABLE gestion__productos_tipos
  ADD COLUMN cont_cuenta_compra_default_id INT(11) DEFAULT NULL,
  ADD COLUMN cont_cuenta_venta_default_id INT(11) DEFAULT NULL;

UPDATE gestion__productos_tipos
SET cont_cuenta_compra_default_id = 131,  -- Mercaderias
    cont_cuenta_venta_default_id = 600    -- Ventas Productos
WHERE maneja_stock = 1; -- tipo 'Producto'

UPDATE gestion__productos_tipos
SET cont_cuenta_venta_default_id = 601    -- Ventas Servicios
    -- cont_cuenta_compra_default_id de Servicio queda NULL a proposito:
    -- la cuenta de gasto varia demasiado producto a producto
    -- (luz, honorarios, flete...) para tener un default unico razonable.
WHERE maneja_stock = 0; -- tipo 'Servicio'

-- ---------------------------------------------------------------------
-- 6) Indice para resolver saldo de cuenta corriente por cliente contra
--    el mayor contable (evita tabla de saldo separada — ver nota).
--    (Sin cambios respecto a v1.0.)
-- ---------------------------------------------------------------------
ALTER TABLE gestion__cont_asientos_detalles
  ADD KEY idx_entidad_cuenta (entidad_id, cuenta_id);

COMMIT;
